// services/suppliers/bidfoodParser.js

const { extractInvoiceDate } = require("./extractInvoiceMeta");

function normalizeText(text) {
  return text
    .replace(/\r\n/g, "\n")
    .replace(/[|]+/g, " ")
    .replace(/[^\S\n]+/g, " ")
    .replace(/\s{2,}/g, " ")
    .replace(/[^\x20-\x7E\n]/g, "")
    .trim();
}

// Extract Document No (Bidfood invoice number)
function extractBidfoodDocumentNumber(text) {
  const patterns = [
    /Document\s*No[:\s]*([A-Z0-9.\-]+)/i,
    /Document\s*Number[:\s]*([A-Z0-9.\-]+)/i
  ];

  for (const re of patterns) {
    const m = text.match(re);
    if (m && m[1]) return m[1].trim();
  }

  return "";
}

function isValidBidfoodCode(code) {
  return /^\d{3,6}$/.test(code.trim());
}

function parseItems(lines) {
  const items = [];

  // Find the header row "Code Product Description Brand Pack Size ..."
  let startIndex = lines.findIndex(l =>
    /Code\s+Product\s+Description/i.test(l)
  );

  if (startIndex === -1) return items;

  // Start reading items after header
  for (let i = startIndex + 1; i < lines.length; i++) {
    const raw = lines[i].trim();
    if (!raw) continue;

    // Stop at "INVOICE TOTAL"
    if (/INVOICE\s+TOTAL/i.test(raw)) break;

    const tokens = raw.split(/\s+/);
    if (tokens.length < 3) continue;

    const code = tokens.shift();

    if (!isValidBidfoodCode(code)) continue;

    // Now extract fields based on Bidfood table structure
    // Code | Description | Brand | Pack Size | UOM | Qty | Unit Price | Price | Excl GST | GST | Total

    let description = "";
    let brand = "";
    let packSize = "";
    let uom = "";
    let qty = 0;
    let unitPrice = 0;
    let total = 0;

    // Description = until we hit brand (uppercase words)
    let idx = 0;

    // Description: read until we hit a token that looks like brand or pack size
    while (idx < tokens.length && !/^[A-Z]{2,}$/.test(tokens[idx]) && !/\d/.test(tokens[idx])) {
      description += tokens[idx] + " ";
      idx++;
    }
    description = description.trim();

    // Brand
    if (idx < tokens.length && /^[A-Z]{2,}$/.test(tokens[idx])) {
      brand = tokens[idx];
      idx++;
    }

    // Pack size (e.g., "1kg", "2kg", "500g")
    if (idx < tokens.length && /\d/.test(tokens[idx])) {
      packSize = tokens[idx];
      idx++;
    }

    // Unit of measure (PKT, CTN, BAG, etc.)
    if (idx < tokens.length && /^[A-Z]{2,5}$/i.test(tokens[idx])) {
      uom = tokens[idx];
      idx++;
    }

    // Quantity
    if (idx < tokens.length && /^\d+(\.\d+)?$/.test(tokens[idx])) {
      qty = parseFloat(tokens[idx]);
      idx++;
    }

    // Unit price
    if (idx < tokens.length && /^\d+(\.\d+)?$/.test(tokens[idx])) {
      unitPrice = parseFloat(tokens[idx]);
      idx++;
    }

    // Skip price/excl GST/GST
    while (idx < tokens.length - 1) idx++;

    // Total value (last number)
    if (idx < tokens.length && /^\d+(\.\d+)?$/.test(tokens[idx])) {
      total = parseFloat(tokens[idx]);
    }

    items.push({
      product_code: code,
      description,
      brand,
      pack_size: packSize,
      unit: uom,
      quantity: qty,
      unit_price: unitPrice,
      gst: 0,
      total
    });
  }

  return items;
}

module.exports = function parseBidfoodInvoice(ocrText) {
  const text = normalizeText(ocrText);
  const lines = text.split("\n").map(l => l.trim()).filter(Boolean);

  const invoiceNumber = extractBidfoodDocumentNumber(text);
  const invoiceDate = extractInvoiceDate(text);
  const items = parseItems(lines);

  return {
    supplier: "BIDFOOD",
    invoiceNumber,
    invoiceDate,
    items
  };
};

function clean(val) {
  if (!val) return 0;
  return Number(
    val
      .replace(/i/gi, "1")
      .replace(/l/gi, "1")
      .replace(/o/gi, "0")
      .replace(/H/gi, "0")
      .replace(/[^\d.]/g, "")
  );
}

function parsePfd(text) {
  const lines = text.split("\n").map(l => l.trim()).filter(Boolean);
  const flat = text.replace(/\n/g, " ").replace(/\s+/g, " ");

  // Invoice number
  let invoiceNumber =
    (flat.match(/Invoice[:\s]+([A-Za-z0-9]+)/i) || [])[1] ||
    (flat.match(/Invoice\s*No[:\s]+([A-Za-z0-9]+)/i) || [])[1] ||
    (flat.match(/imvoicaNo[:\s]+([A-Za-z0-9]+)/i) || [])[1] ||
    null;

  // Invoice date
  let invoiceDate =
    (flat.match(/Invoice\s*Date[:\s]+([0-9]{1,2}\s+[A-Za-z]{3}\s+[0-9]{2,4})/i) || [])[1] ||
    null;

  const items = [];

  for (const line of lines) {
    // Product code
    const codeMatch = line.match(/^(\d{5,6})/);
    if (!codeMatch) continue;
    const code = codeMatch[1];

    // Description
    const descMatch = line.match(/^\d+\]?\s+(.*?)\s+\[/);
    const description = descMatch ? descMatch[1].trim() : "";

    // Extract the bracket block
    const blockMatch = line.match(/\[(.*?)\]/);
    if (!blockMatch) continue;

    const block = blockMatch[1];

    // Extract numbers inside the block
    let nums = block.match(/[0-9]+\.[0-9]+|[0-9]+/g) || [];

    // Clean numbers
    nums = nums.map(clean);

    // Now extract trailing GST + total
    const tailMatch = line.match(/\]\s*([0-9]+\.[0-9]+)\s*\|\s*([0-9]+\.[0-9]+)/);
    const gst = tailMatch ? clean(tailMatch[1]) : 0;
    const total = tailMatch ? clean(tailMatch[2]) : 0;

    // Assign fields
    const qtyOrdered = nums[0] || 1;
    const qtySupplied = nums[1] || 1;
    const price = nums[2] || 0;
    const netValue = nums[3] || price;

    items.push({
      code,
      description,
      unit: "", // UM optional
      qty_ordered: qtyOrdered,
      qty_supplied: qtySupplied,
      unit_price: price,
      net_value: netValue,
      gst,
      total
    });
  }

  return {
    supplier: "PFD",
    invoiceNumber,
    invoiceDate,
    items
  };
}

module.exports = parsePfd;
